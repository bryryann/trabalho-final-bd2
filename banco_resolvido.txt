CREATE TABLE Usuario ( 
    id_usuario NUMBER PRIMARY KEY,  
    email VARCHAR2(50),  
    senha VARCHAR2(50),  
    nome_usuario VARCHAR2(50),  
    plano VARCHAR2(50) DEFAULT 'Free',  
    telefone VARCHAR2(20)
);

CREATE TABLE Inventario ( 
    id_inventario NUMBER PRIMARY KEY,  
    tamanho NUMBER,  
    peso NUMBER  
); 

CREATE TABLE Player ( 
    id_player NUMBER PRIMARY KEY,  
    id_usuario NUMBER,  
    id_inventario NUMBER,  
    level_player NUMBER,  
    nome_player VARCHAR2(50),  
    experiencia NUMBER,  
    kills NUMBER,  
    faccao VARCHAR2(10),  
    deaths NUMBER,  
    rublo NUMBER,  
    dolar NUMBER,  
    euro NUMBER,
    CONSTRAINT fk_player_usuario FOREIGN KEY (id_usuario) REFERENCES Usuario(id_usuario),
    CONSTRAINT fk_player_inventario FOREIGN KEY (id_inventario) REFERENCES Inventario(id_inventario)
); 

CREATE TABLE Item ( 
    id_item NUMBER PRIMARY KEY,  
    tamanho_item NUMBER,  
    classe_item VARCHAR2(50),  
    peso_item NUMBER
); 

CREATE TABLE Mapa ( 
    id_mapa NUMBER PRIMARY KEY,  
    nome_mapa VARCHAR2(50),  
    descricao_mapa VARCHAR2(200),  
    tamanho_mapa VARCHAR2(20),  
    duracao NUMBER
);

CREATE TABLE Raid (
    id_raid NUMBER PRIMARY KEY,
    id_mapa NUMBER,
    inicio_raid TIMESTAMP,
    horario_raid TIMESTAMP,
    status_raid VARCHAR2(50),
    CONSTRAINT fk_raid_mapa FOREIGN KEY (id_mapa) REFERENCES Mapa(id_mapa)
);

CREATE TABLE Banimento ( 
    id_banimento NUMBER PRIMARY KEY,  
    id_usuario NUMBER,  
    descricao VARCHAR2(50),  
    data_banimento DATE,  
    data_desbanimento DATE,
    CONSTRAINT fk_banimento_usuario FOREIGN KEY (id_usuario) REFERENCES Usuario(id_usuario)
); 

CREATE TABLE Player_Raid ( 
    id_player_raid NUMBER PRIMARY KEY,
    id_player NUMBER,  
    id_raid NUMBER,  
    id_morto_por NUMBER,  
    status_player VARCHAR2(10) DEFAULT 'IN ACTION',  
    scav NUMBER(1),  
    dano_causado NUMBER,
    CONSTRAINT fk_playerraid_player FOREIGN KEY (id_player) REFERENCES Player(id_player),
    CONSTRAINT fk_playerraid_raid FOREIGN KEY (id_raid) REFERENCES Raid(id_raid)
); 

CREATE TABLE Instancia_item ( 
    id_instancia_item NUMBER PRIMARY KEY,
    id_item NUMBER,
    id_inventario NUMBER,
    in_raid NUMBER(1),
    durabilidade NUMBER,
    in_container_seguro NUMBER(1),
    quantidade NUMBER,
    CONSTRAINT fk_instancia_item FOREIGN KEY (id_item) REFERENCES Item(id_item),
    CONSTRAINT fk_instancia_inv FOREIGN KEY (id_inventario) REFERENCES Inventario(id_inventario)
);
